cmake_minimum_required(VERSION 3.8)
project(tianrover)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2_msgs REQUIRED)
add_executable(tf_authority_guard src/tf_authority_guard.cpp)
ament_target_dependencies(tf_authority_guard rclcpp std_msgs tf2_msgs)
install(TARGETS tf_authority_guard DESTINATION lib/${PROJECT_NAME})
ament_python_install_package(${PROJECT_NAME})
ament_environment_hooks(env-hooks/99.tianrover.sh)
install(PROGRAMS scripts/ekf_inputs DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/lidar_cloud_gate scripts/lidar_velocity DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/cartographer_map_save scripts/map_save_3d DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config urdf patches DESTINATION share/${PROJECT_NAME}
  PATTERN "__pycache__" EXCLUDE)
install(FILES README.md TODO.md VALIDATION.md MODEL.md EKF.md THREED.md VALIDATION_3D.md DEPENDENCIES_3D.md dependencies-3d.repos dependencies-2d.repos SLAM_2D.md LICENSE LICENSE.nav2 NOTICE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  find_package(launch_testing_ament_cmake REQUIRED)
  ament_add_pytest_test(config_contract test/test_config.py)
  ament_add_pytest_test(mars_model test/test_mars_urdf.py)
  ament_add_pytest_test(ekf_inputs test/test_ekf_inputs.py)
  ament_add_pytest_test(map_database test/test_map_database.py)
  ament_add_pytest_test(lidar_frontend test/test_lidar_frontend.py)
  add_launch_test(test/test_runtime.launch.py TIMEOUT 300)
  add_launch_test(test/test_slam_2d.launch.py TIMEOUT 300)
  add_launch_test(test/test_ekf.launch.py TIMEOUT 300)
  add_launch_test(test/test_kiss.launch.py TIMEOUT 150)
  add_launch_test(test/test_rtabmap.launch.py TIMEOUT 330)
  add_launch_test(test/test_tf_guard.launch.py TIMEOUT 120)
endif()
ament_package()
